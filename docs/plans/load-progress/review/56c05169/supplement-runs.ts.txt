interface Page {
  sessionId: string;
  targetId: string;
  evalv: <T = unknown>(expr: string) => Promise<T | undefined>;
  shot: (name: string, note?: string) => Promise<unknown>;
  close: () => Promise<void>;
}

const measurements: Record<string, unknown> = {};

async function openPage(
  url: string,
  cfg: Cfg,
  opts: { cpu?: number; disableScripts?: boolean; failPatterns?: string[]; navigate?: boolean } = {}
): Promise<Page> {
  const created = (await cdp.send('Target.createTarget', { url: 'about:blank' })) as { targetId: string };
  const targetId = created.targetId;
  const attached = (await cdp.send('Target.attachToTarget', { targetId, flatten: true })) as { sessionId: string };
  const sessionId = attached.sessionId;
  await cdp.send('Page.enable', {}, sessionId);
  await cdp.send('Runtime.enable', {}, sessionId);
  await cdp.send('Network.enable', {}, sessionId);
  await cdp.send('Network.setCacheDisabled', { cacheDisabled: true }, sessionId);
  await cdp.send('Emulation.setDeviceMetricsOverride', { width: cfg.viewport.width, height: cfg.viewport.height, deviceScaleFactor: 1, mobile: cfg.viewport.mobile ?? false }, sessionId);
  await cdp.send('Emulation.setEmulatedMedia', { media: 'screen', features: [{ name: 'prefers-color-scheme', value: cfg.colorScheme }] }, sessionId);
  if (opts.cpu && opts.cpu > 1) await cdp.send('Emulation.setCPUThrottlingRate', { rate: opts.cpu }, sessionId);
  if (opts.disableScripts) await cdp.send('Emulation.setScriptExecutionDisabled', { value: true }, sessionId);
  let stop: (() => void) | undefined;
  if (opts.failPatterns && opts.failPatterns.length > 0) {
    await cdp.send('Fetch.enable', { patterns: opts.failPatterns.map((p) => ({ urlPattern: p, requestStage: 'Request' })) }, sessionId);
    stop = cdp.onEvent((e) => {
      if (e.method === 'Fetch.requestPaused' && e.sessionId === sessionId) {
        const reqId = (e.params as { requestId: string }).requestId;
        void cdp.send('Fetch.failRequest', { requestId: reqId, errorReason: 'ConnectionRefused' }, sessionId).catch(() => {});
      }
    });
  }
  const evalv = async <T,>(expr: string): Promise<T | undefined> => {
    const res = (await cdp.send('Runtime.evaluate', { expression: expr, returnByValue: true, awaitPromise: true }, sessionId).catch(() => null)) as { result?: { value?: T } } | null;
    return res?.result?.value;
  };
  const shot = async (name: string, note?: string) => {
    const before = await evalv(MEASURE);
    const s = (await cdp.send('Page.captureScreenshot', { format: 'png' }, sessionId).catch(() => null)) as { data: string } | null;
    const after = await evalv(MEASURE);
    if (s) await writeFile(join(OUTX, `${name}.png`), Buffer.from(s.data, 'base64'));
    measurements[name] = { note, before, after };
    console.log(`  shot ${name}`);
    return after;
  };
  if (opts.navigate !== false) await cdp.send('Page.navigate', { url }, sessionId);
  return {
    sessionId,
    targetId,
    evalv,
    shot,
    close: async () => {
      stop?.();
      await cdp.send('Target.closeTarget', { targetId }).catch(() => {});
    },
  };
}

async function waitFor(page: Page, expr: string, timeoutMs = 20_000, stepMs = 4): Promise<boolean> {
  const t = Date.now();
  while (Date.now() - t < timeoutMs) {
    if (await page.evalv<boolean>(expr)) return true;
    await Bun.sleep(stepMs);
  }
  return false;
}

      await cdp.send('Emulation.setEmulatedMedia', { media: 'screen', features: [{ name: 'prefers-color-scheme', value: 'light' }, { name: 'prefers-reduced-motion', value: 'reduce' }] }, page.sessionId);
      await cdp.send('Page.navigate', { url: html.shareUrl }, page.sessionId);
      if (await waitFor(page, PILL_VISIBLE, 30_000, 2)) await page.shot('G-pill-reduced-motion-html-1440-light');
      await page.close();
    }
  }

  // F. JSX under a sandbox CSP that allows blob: (as app.ts serveSandbox does).
  if (want('jsx')) {
    const jsx = byKey('jsx');
    const page = await openPage(jsx.shareUrl, C1440L, {});
    await waitFor(page, READY, 20_000, 20);
    await Bun.sleep(400);
    await page.shot('F-jsx-ready-blob-csp-1440-light');
    await page.close();
  }

  await writeFile(join(OUTX, 'measurements.json'), JSON.stringify(measurements, null, 2));
  console.log('REVIEW_CAPTURE_DONE');
} finally {
  child.kill('SIGKILL');
  service.stop(true);
  usercontentService.stop(true);
  await rm(profileDir, { recursive: true, force: true }).catch(() => {});
  await rm(scratch, { recursive: true, force: true }).catch(() => {});
}
